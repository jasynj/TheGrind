class Solution:
    def minSubArrayLen(self, target: int, nums: list[int]) -> int:
        
        left = 0
        total = 0
        
        lowest = float('inf')
        for right in range(len(nums)): #3
            total += nums[right] #total = 10
             #lowest = 1

            while total >= target:
                lowest = min(right - left + 1, lowest)
                total -= nums[left]
                left += 1

        return 0 if lowest == float('inf') else lowest
