class Solution:
    def mergeAlternately(self, word1: str, word2: str) -> str:
        p = q = 0
        res = []

        while p < len(word1) or q < len(word2):

            if p < len(word1):
                res.append(word1[p])
                p += 1

            if q < len(word2):
                res.append(word2[q])
                q += 1

        return "".join(res)
