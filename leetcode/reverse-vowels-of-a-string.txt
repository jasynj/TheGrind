        while l < r:
            while l < r and ss[l].lower() not in vowels:
                l += 1
            
            while l < r and ss[r].lower() not in vowels:
        ss = list(s)
        vowels = ['a', 'e', 'i','o', 'u']
                r -= 1

            ss[l], ss[r] = ss[r], ss[l]
            l += 1
            r -= 1

        return "".join(ss)
