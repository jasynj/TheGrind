class Solution:
    def findDifference(self, nums1: List[int], nums2: List[int]) -> List[List[int]]:
        num1 = []
        
        for i in set(nums1):
            if i not in set(nums2):
                num1.append(i)

        for i in set(nums2):
            if i not in set(nums1):
        num2 = []
                num2.append(i)

        return [num1, num2]
