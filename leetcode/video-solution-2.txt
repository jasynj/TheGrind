class Solution:
    def mergeAlternately(self, word1: str, word2: str) 
-> str:
        m,n = len(word1), len(word2)
        merged = []

        if m > n:
            for i in range(n):
                merged.append(word1[i])
                merged.append(word2[i])

            merged.append(word1[n:])
